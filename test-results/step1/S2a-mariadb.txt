SQL generato (toString(true)):
SELECT
	libri.`titolo` AS libri_titolo,
	editori.`nome` AS editori_nome,
	COUNT(autori.`id`) AS n_autori
FROM
	`editori` editori
	INNER JOIN `libri` libri
	 ON editori.`id` = libri.`id_editore`
	INNER JOIN `libri_autori` libri_autori
	 ON libri.`id` = libri_autori.`id_libro`
	INNER JOIN `autori` autori
	 ON libri_autori.`id_autore` = autori.`id`
WHERE
	libri.`anno` > 1960
GROUP BY
	libri.`titolo`,
	editori.`nome`
ORDER BY
	editori.`nome` DESC,
	libri.`titolo` ASC

SQL di riferimento scritto a mano:
SELECT l.titolo, e.nome, COUNT(a.id) AS n_autori FROM libri l JOIN editori e ON l.id_editore = e.id JOIN libri_autori la ON la.id_libro = l.id JOIN autori a ON la.id_autore = a.id WHERE l.anno > 1960 GROUP BY l.titolo, e.nome ORDER BY e.nome DESC, l.titolo ASC
